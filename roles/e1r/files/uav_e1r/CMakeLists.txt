cmake_minimum_required(VERSION 3.16)
project(uav_e1r)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
set(RS_DRIVER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/rs_driver" CACHE PATH "Pinned official RoboSense driver")
if(NOT EXISTS "${RS_DRIVER_ROOT}/src/rs_driver/driver/decoder/decoder_RSE1.hpp")
  message(FATAL_ERROR "Pinned rs_driver checkout missing: ${RS_DRIVER_ROOT}")
endif()
add_executable(e1r_node src/e1r_node.cpp)
target_include_directories(e1r_node PRIVATE include "${RS_DRIVER_ROOT}/src")
target_compile_options(e1r_node PRIVATE -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(e1r_node nlohmann_json::nlohmann_json)
ament_target_dependencies(e1r_node rclcpp sensor_msgs std_msgs diagnostic_msgs)
install(TARGETS e1r_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_time_gate test/test_time_gate.cpp)
  target_include_directories(test_time_gate PRIVATE include)
  target_compile_options(test_time_gate PRIVATE -UNDEBUG)
  add_test(NAME e1r_time_gate COMMAND test_time_gate)
  add_executable(test_vendor_decoder test/test_vendor_decoder.cpp)
  target_include_directories(test_vendor_decoder PRIVATE include "${RS_DRIVER_ROOT}/src")
  target_compile_options(test_vendor_decoder PRIVATE -UNDEBUG)
  add_test(NAME e1r_vendor_decoder COMMAND test_vendor_decoder)
endif()
ament_package()
