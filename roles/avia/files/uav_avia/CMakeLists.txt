cmake_minimum_required(VERSION 3.16)
project(uav_avia)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(livox_ros_driver2 REQUIRED)
find_package(Threads REQUIRED)
set(LIVOX_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/Livox-SDK" CACHE PATH "Pinned official SDK1 checkout")
add_subdirectory("${LIVOX_SDK_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/livox" EXCLUDE_FROM_ALL)
# Keep the vendor's original C++11 build contract, independent of ROS Jazzy C++17.
set_target_properties(livox_sdk_static PROPERTIES CXX_STANDARD 11)
# Bundled fmt names an enum char8_t, legal in C++11; GCC13 warns about C++20.
# Scope this compatibility exception to the untouched legacy SDK target.
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  target_compile_options(livox_sdk_static PRIVATE -Wno-c++20-compat)
endif()
add_executable(avia_node src/avia_node.cpp)
target_include_directories(avia_node PRIVATE include)
target_compile_options(avia_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(avia_node livox_sdk_static Threads::Threads)
ament_target_dependencies(avia_node rclcpp sensor_msgs diagnostic_msgs livox_ros_driver2)
install(TARGETS avia_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_point test/test_point.cpp)
  target_include_directories(test_point PRIVATE include)
  target_compile_options(test_point PRIVATE -UNDEBUG)
  add_test(NAME avia_point_contract COMMAND test_point)
  add_executable(test_time test/test_time.cpp)
  target_include_directories(test_time PRIVATE include)
  target_compile_options(test_time PRIVATE -UNDEBUG)
  add_test(NAME avia_utc_contract COMMAND test_time)
endif()
ament_package()
